import { createHash } from "node:crypto";
import { verifyFilesPrincipalBinding } from "./files-principal-binding";
import { runCauseEffect } from "../../backend/effect-boundary";
import type { TeamApiAgents } from "./dependencies";
import { HttpError } from "./http-error";
import type { TeamApiRequestContext, RouteOutcome } from "./request-context";
import { readJson, requireListedAgent, requireAgentLevel, memberSender, stringField, stringArray, nullableString, readBinary } from "./request-helpers";

/** Exact existing main/task send. The negotiated codec is mandatory; there is no legacy fallback. */
export async function routeSledFilesIdentity(context: TeamApiRequestContext, agents: TeamApiAgents, hiddenAgentIds: ReadonlySet<string>, trusted?: { secret: string; hostId: string; environmentId: string }): Promise<RouteOutcome> {
  if (context.method !== "POST" || context.url.pathname !== "/v1/sled/files/send") return "unmatched";
  if (!context.capabilities.has("sled-files-identity-v1")) throw new HttpError(400, "Guarded Files unavailable.");
  const signed = context.request.headers["x-sled-files-principal"];
  const raw = signed === undefined ? null : await readBinary(context.request, 120000);
  const body = raw === null ? await readJson(context.request) : JSON.parse(Buffer.from(raw).toString("utf8"));
  let principal: string | undefined;
  if (raw !== null) {
    if (!context.capabilities.has("sled-files-upload-recovery-v1") || !trusted || typeof signed !== "string" || body.environmentId !== trusted.environmentId) throw new HttpError(403, "Original draft owner unavailable");
    principal = verifyFilesPrincipalBinding(trusted.secret, signed, { hostId: trusted.hostId, method: "POST", path: "/v1/sled/files/send", bodySha256: createHash("sha256").update(raw).digest("hex") }, Date.now());
  }
  const agentId = stringField(body, "agentId");
  if (hiddenAgentIds.has(agentId)) throw new HttpError(404, "Agent unavailable.");
  requireListedAgent(agents, agentId);
  requireAgentLevel(context, agentId, "chat");
  if (typeof body.receiptOnly !== "boolean") throw new HttpError(400, "Invalid receipt lookup.");
  const taskId = nullableString(body, "taskId");
  return context.json(202, await runCauseEffect(agents.sendMessage({
    ...(principal === undefined ? {} : { sledUploadOwner: { principal, environmentId: trusted!.environmentId, hostId: trusted!.hostId, agentId, threadId: stringField(body, "expectedThreadId"), taskId } }),
    agentId, expectedThreadId: stringField(body, "expectedThreadId"),
    clientMessageId: stringField(body, "clientMessageId"), receiptOnly: body.receiptOnly,
    text: stringField(body, "text", true, 100_000), attachmentDraftIds: stringArray(body, "attachmentDraftIds"),
    replyToMessageId: nullableString(body, "replyToMessageId"), ...(taskId === null ? {} : { taskId }),
  }, memberSender(context.member))));
}
