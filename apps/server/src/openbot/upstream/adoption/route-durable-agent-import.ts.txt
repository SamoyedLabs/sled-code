import { isDynamicRecord } from "@openbot/contracts/runtime-values";
import { answer } from "./route-agent-import";
import type { DurableAgentImport } from "../durable-agent-import";
import { HttpError } from "./http-error";
import type { RouteOutcome, TeamApiRequestContext } from "./request-context";
import { readBinary, readJson, stringField } from "./request-helpers";
import { AGENT_IMPORT_UPLOAD_BYTES } from "@openbot/contracts/team-protocol/agent-import-v1";
export const SLED_IMPORT_CAPABILITY = "sled-agent-import-v1";
export async function routeDurableAgentImport(context: TeamApiRequestContext, service: DurableAgentImport | undefined, newAgentHidden: () => boolean, hidden: ReadonlySet<string> = new Set()): Promise<RouteOutcome> {
  const { method, url, member, capabilities, request, json } = context;
  const operation = new Map([
    ["/v1/sled-agent-import/stage", "stage"], ["/v1/sled-agent-import/status", "status"],
    ["/v1/sled-agent-import/apply", "apply"], ["/v1/sled-agent-import/discard", "discard"],
  ]).get(url.pathname);
  if (method !== "POST" || !operation) return "unmatched";
  if (!service || !capabilities.has(SLED_IMPORT_CAPABILITY)) throw new HttpError(400, "Durable import unavailable");
  const owner = member.id;
  const respond = (receipt: Awaited<ReturnType<DurableAgentImport["status"]>>) => {
    const result = receipt?.result;
    if (isDynamicRecord(result) && Array.isArray(result.agents) && result.agents.some(agent => isDynamicRecord(agent) && typeof agent.agentId === "string" && (hidden.has(agent.agentId) || context.agentLevel(agent.agentId) === "none"))) throw new HttpError(404, "Import unavailable");
    return json(200, receipt);
  };
  if (operation === "status") return respond(await service.status(owner));
  if (operation === "stage") {
    return respond(await answer(() => service.stageUpload(owner, url.searchParams.get("requestId") ?? "", () => readBinary(request, AGENT_IMPORT_UPLOAD_BYTES)), 400));
  }
  const body = await readJson(request);
  if (operation === "discard") return respond(await answer(() => service.discard(owner, stringField(body, "token")), 409));
  if (newAgentHidden()) throw new HttpError(400, "New agent provider is local-only");
  return respond(await answer(() => service.apply(owner, body, { id: member.id, name: member.name ?? "Team member" }), 409));
}
